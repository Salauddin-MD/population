\chapter{Simple species Population Model}
\section*{Important Info:}
\begin{itemize}
    \item Malthusian model:
    $$\frac{dN}{dt} = kN$$
    \item solution of M model:
    $$ N(t) = N_0 e^{kt}$$
    \item Logistic Model:
     $$\frac{dN}{dt} = r \left(1 - \frac{N}{K}\right) N $$
     Solution:
     $$ N = \frac{K N_0}{N_0 + (K - N_0) e^{-rt}}$$
\end{itemize}
